#include <Arduino.h>
#include <TinyGPSPlus.h>

#define RXD2 16
#define TXD2 17

TinyGPSPlus gps;
HardwareSerial gpsSerial(2);

void setup() {
  Serial.begin(115200);
  gpsSerial.begin(9600, SERIAL_8N1, RXD2, TXD2);
  
  Serial.println("=== NEO-6M GPS Test Starting ===");
  Serial.println("Powering from VIN (5V) - Should work");
  Serial.println("Take the board OUTDOORS now!");
}

void loop() {
  while (gpsSerial.available() > 0) {
    char c = gpsSerial.read();
    gps.encode(c);
    // Serial.write(c);        // Uncomment only if you want raw data
  }

  static unsigned long lastUpdate = 0;
  if (millis() - lastUpdate > 2000) {   // Print every 2 seconds
    lastUpdate = millis();

    Serial.printf("Satellites in view: %d | ", gps.satellites.value());
    
    if (gps.location.isValid()) {
      Serial.printf("✅ FIX! Lat: %.6f Lon: %.6f | Alt: %.1fm\n",
                    gps.location.lat(), gps.location.lng(), gps.altitude.meters());
    } else {
      Serial.println("⏳ No Fix Yet - Keep antenna pointing to sky");
    }
  }
}
